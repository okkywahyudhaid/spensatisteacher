var TZ = 'Asia/Jakarta';
var HARI = ['Senin','Selasa','Rabu','Kamis','Jumat','Sabtu'];

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('Sistem Piket Absensi Guru')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function sh_(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet(), s = ss.getSheetByName(name);
  if (!s) { // cari tanpa membedakan huruf besar/kecil dan spasi di ujung nama
    var all = ss.getSheets(), key = name.toLowerCase();
    for (var i = 0; i < all.length; i++) if (all[i].getName().trim().toLowerCase() === key) { s = all[i]; break; }
  }
  if (!s && name === 'Log_Absensi') {
    s = ss.insertSheet(name);
    s.appendRow(['Timestamp','Tanggal','Hari','Guru_Piket','Kelas','Jam_Ke','Guru_Mapel','Status','Keterangan']);
  }
  if (!s) throw new Error('Sheet "' + name + '" tidak ditemukan. Sheet yang ada: ' + ss.getSheets().map(function (x) { return '"' + x.getName() + '"'; }).join(', '));
  return s;
}
function ymd_(v) { return v instanceof Date ? Utilities.formatDate(v, TZ, 'yyyy-MM-dd') : String(v).substring(0, 10); }

// ===== SESI =====
function login(username, password) {
  var d = sh_('Data_Guru').getDataRange().getValues();
  for (var i = 1; i < d.length; i++) {
    if (String(d[i][0]).toLowerCase() == String(username).trim().toLowerCase() && String(d[i][1]) == String(password)) {
      var user = { username: d[i][0], name: d[i][2], role: String(d[i][3] || 'guru').toLowerCase() };
      var token = Utilities.getUuid();
      CacheService.getScriptCache().put(token, JSON.stringify(user), 21600);
      user.token = token; user.success = true;
      return user;
    }
  }
  return { success: false, message: 'Username atau Password salah!' };
}

function auth_(token, needAdmin) {
  var s = CacheService.getScriptCache().get(token);
  if (!s) throw new Error('Sesi habis, silakan login ulang.');
  var u = JSON.parse(s);
  if (needAdmin && u.role !== 'admin') throw new Error('Hanya admin yang boleh mengakses.');
  return u;
}

function logout(token) { CacheService.getScriptCache().remove(token); }

// ===== DATA UMUM =====
// Daftar guru untuk dropdown Jadwal Pelajaran dan Jadwal Guru Piket.
// Mengambil nama dari kolom C Data_Guru dan tidak menampilkan akun admin.
function getListGuru(token) {
  auth_(token);
  var d = sh_('Data_Guru').getDataRange().getValues();
  var seen = {}, list = [];
  for (var i = 1; i < d.length; i++) {
    var nama = String(d[i][2] || '').trim();
    var role = String(d[i][3] || 'guru').trim().toLowerCase();
    var key = nama.toLowerCase();
    if (nama && role !== 'admin' && !seen[key]) {
      seen[key] = true;
      list.push(nama);
    }
  }
  return list.sort(function(a, b) { return a.localeCompare(b, 'id'); });
}

// Jadwal dengan kode guru sudah diterjemahkan ke nama
function getJadwal(token) {
  auth_(token);
  var g = sh_('Data_Guru').getDataRange().getValues(), map = {};
  for (var i = 1; i < g.length; i++) map[String(g[i][0]).toUpperCase()] = g[i][2];
  var d = sh_('Jadwal').getDataRange().getValues(), out = [];
  for (var j = 1; j < d.length; j++) {
    if (!d[j][0]) continue;
    out.push({ row: j + 1, hari: d[j][0], waktu: d[j][1], jam: Number(d[j][2]), piket: d[j][3], kelas: d[j][4],
      kode: d[j][5] || '', guru: map[String(d[j][5]).toUpperCase()] || '' });
  }
  return out;
}


// ===== ADMIN: KELOLA JADWAL PELAJARAN =====
// Struktur sheet Jadwal yang digunakan aplikasi:
// A Hari | B Waktu | C Jam | D Guru Piket | E Kelas | F Kode_Guru
function simpanJadwal(token, f) {
  try {
    auth_(token, true);
    var hari = String(f.hari || '').trim();
    var waktu = String(f.waktu || '').trim();
    var jam = Number(f.jam);
    var piket = String(f.piket || '').trim();
    var kelas = String(f.kelas || '').trim().toUpperCase();
    var namaGuru = String(f.guru || '').trim();

    if (HARI.indexOf(hari) < 0) return { success: false, message: 'Hari tidak valid.' };
    if (!waktu) return { success: false, message: 'Waktu pelajaran wajib diisi.' };
    if (!Number.isInteger(jam) || jam < 1 || jam > 12) return { success: false, message: 'Jam pelajaran harus berupa angka 1–12.' };
    if (!/^([789])[A-H]$/.test(kelas)) return { success: false, message: 'Kelas tidak valid.' };
    if (!namaGuru) return { success: false, message: 'Guru mata pelajaran wajib dipilih.' };

    var guruData = sh_('Data_Guru').getDataRange().getValues(), kodeGuru = '';
    for (var i = 1; i < guruData.length; i++) {
      if (String(guruData[i][2] || '').trim().toLowerCase() === namaGuru.toLowerCase() &&
          String(guruData[i][3] || 'guru').toLowerCase() !== 'admin') {
        kodeGuru = String(guruData[i][0] || '').trim();
        break;
      }
    }
    if (!kodeGuru) return { success: false, message: 'Guru tidak ditemukan atau akun bukan berstatus guru.' };

    var sheet = sh_('Jadwal'), data = sheet.getDataRange().getValues();
    var row = Number(f.row || 0);
    for (var r = 1; r < data.length; r++) {
      if (r + 1 === row) continue;
      if (String(data[r][0]).trim() === hari &&
          String(data[r][4]).trim().toUpperCase() === kelas &&
          Number(data[r][2]) === jam) {
        return { success: false, message: 'Jadwal untuk ' + hari + ', kelas ' + kelas + ', jam ke-' + jam + ' sudah ada.' };
      }
    }
    var values = [[hari, waktu, jam, piket, kelas, kodeGuru]];
    if (row >= 2 && row <= sheet.getLastRow()) {
      // Jadwal pelajaran tidak mengubah kolom guru piket.
      values[0][3] = data[row - 1][3];
      sheet.getRange(row, 1, 1, 6).setValues(values);
      return { success: true, message: 'Jadwal ' + hari + ' kelas ' + kelas + ' jam ke-' + jam + ' berhasil diperbarui.' };
    }
    sheet.appendRow(values[0]);
    return { success: true, message: 'Jadwal baru berhasil ditambahkan.' };
  } catch (e) {
    return { success: false, message: e.message };
  }
}

// ===== ADMIN: UBAH GURU PIKET PER SLOT WAKTU =====
function simpanGuruPiket(token, f) {
  try {
    auth_(token, true);
    f = f || {};
    var hari = String(f.hari || '').trim();
    var waktu = String(f.waktu || '').trim();
    var jam = Number(f.jam);
    var namaGuru = String(f.guru || '').trim();

    if (HARI.indexOf(hari) < 0) return { success: false, message: 'Hari tidak valid.' };
    if (!waktu) return { success: false, message: 'Waktu pelajaran harus dipilih.' };
    if (!Number.isInteger(jam) || jam < 1 || jam > 12)
      return { success: false, message: 'Jam pelajaran tidak valid.' };
    if (!namaGuru) return { success: false, message: 'Silakan pilih guru piket.' };

    // Pastikan nama yang dipilih benar-benar berasal dari Data_Guru dan bukan akun admin.
    var guruData = sh_('Data_Guru').getDataRange().getValues();
    var valid = false;
    for (var i = 1; i < guruData.length; i++) {
      var nama = String(guruData[i][2] || '').trim();
      var role = String(guruData[i][3] || 'guru').trim().toLowerCase();
      if (nama.toLowerCase() === namaGuru.toLowerCase() && role !== 'admin') {
        valid = true;
        break;
      }
    }
    if (!valid) return { success: false, message: 'Guru tidak ditemukan dalam daftar guru.' };

    // Satu slot dapat memiliki beberapa baris kelas. Hanya kolom D (Guru Piket)
    // yang diubah; kolom A-C dan E-F (jadwal pelajaran) tetap dipertahankan.
    var sheet = sh_('Jadwal');
    var data = sheet.getDataRange().getValues();
    var rows = [];
    for (var r = 1; r < data.length; r++) {
      var cocok = String(data[r][0] || '').trim() === hari &&
        String(data[r][1] || '').trim() === waktu &&
        Number(data[r][2]) === jam;
      if (cocok) rows.push(r + 1);
    }
    if (!rows.length)
      return { success: false, message: 'Slot jadwal tidak ditemukan. Periksa hari, waktu, dan jam yang dipilih.' };

    rows.forEach(function(row) {
      sheet.getRange(row, 4).setValue(namaGuru);
    });

    return {
      success: true,
      message: 'Guru piket berhasil diperbarui untuk ' + hari + ', jam ke-' + jam + ' (' + rows.length + ' kelas).'
    };
  } catch (e) {
    return { success: false, message: e.message };
  }
}

function hapusJadwal(token, row) {
  try {
    auth_(token, true);
    row = Number(row);
    var sheet = sh_('Jadwal');
    if (!Number.isInteger(row) || row < 2 || row > sheet.getLastRow())
      return { success: false, message: 'Baris jadwal tidak valid.' };
    sheet.deleteRow(row);
    return { success: true, message: 'Jadwal berhasil dihapus.' };
  } catch (e) {
    return { success: false, message: e.message };
  }
}

// ===== GURU: ISI KEHADIRAN =====
function simpanAbsensi(token, f) {
  try {
    var u = auth_(token);
    var sheet = sh_('Log_Absensi'), d = sheet.getDataRange().getValues();
    for (var i = 1; i < d.length; i++) {
      if (ymd_(d[i][1]) === f.tanggal && d[i][4] == f.kelas && Number(d[i][5]) === Number(f.jamKe)) {
        sheet.getRange(i + 1, 1, 1, 9).setValues([[new Date(), f.tanggal, f.hari, u.name, f.kelas, Number(f.jamKe), f.guruMapel, f.status, f.keterangan]]);
        return { success: true, message: 'Data kelas ' + f.kelas + ' jam ke-' + f.jamKe + ' diperbarui.' };
      }
    }
    sheet.appendRow([new Date(), f.tanggal, f.hari, u.name, f.kelas, Number(f.jamKe), f.guruMapel, f.status, f.keterangan]);
    return { success: true, message: 'Data absensi berhasil disimpan!' };
  } catch (e) { return { success: false, message: e.message }; }
}

// Status yang sudah terisi pada tanggal tertentu (untuk penanda di form)
function getTerisi(token, tanggal) {
  auth_(token);
  var d = sh_('Log_Absensi').getDataRange().getValues(), o = {};
  for (var i = 1; i < d.length; i++) if (ymd_(d[i][1]) === tanggal) o[d[i][4] + '|' + Number(d[i][5])] = d[i][7];
  return o;
}

// ===== ADMIN: DASHBOARD MINGGUAN =====
// senin = 'yyyy-MM-dd' (hari Senin pada minggu yang dipilih)
function getDashboard(token, senin) {
  auth_(token, true);
  var p = senin.split('-'), base = new Date(+p[0], +p[1] - 1, +p[2]);
  var tgl = {};
  HARI.forEach(function (h, i) { var x = new Date(base); x.setDate(x.getDate() + i); tgl[h] = Utilities.formatDate(x, TZ, 'yyyy-MM-dd'); });
  var valid = {}; for (var h in tgl) valid[tgl[h]] = 1;
  var log = {}, d = sh_('Log_Absensi').getDataRange().getValues();
  for (var i = 1; i < d.length; i++) {
    var t = ymd_(d[i][1]);
    if (valid[t]) log[t + '|' + d[i][4] + '|' + Number(d[i][5])] = { s: d[i][7], k: d[i][8], m: d[i][6] };
  }
  var jd = getJadwal(token), cells = jd.map(function (r) {
    var t = tgl[r.hari], l = log[t + '|' + r.kelas + '|' + r.jam] || {};
    return { hari: r.hari, waktu: r.waktu, jam: r.jam, piket: r.piket, kelas: r.kelas, kode: r.kode, guru: r.guru, tanggal: t, s: l.s || '', k: l.k || '' };
  });
  return { tanggal: tgl, cells: cells };
}

// ===== ADMIN: REKAP KEHADIRAN PER GURU (MINGGUAN) =====
// Daftar guru diambil dari Data_Guru (akun admin tidak ikut). Jumlah per status
// dihitung dari jadwal minggu terpilih + Log_Absensi (lewat getDashboard).
function getRekapGuru(token, senin) {
  auth_(token, true);
  var dash = getDashboard(token, senin);
  var g = sh_('Data_Guru').getDataRange().getValues(), map = {}, rows = [];
  for (var i = 1; i < g.length; i++) {
    var kode = String(g[i][0] || '').trim();
    if (!kode) continue;
    if (String(g[i][3] || 'guru').trim().toLowerCase() === 'admin') continue;
    var r = { username: kode, nama: String(g[i][2] || kode), total: 0, H: 0, I: 0, S: 0, DL: 0, Cuti: 0, A: 0, belum: 0 };
    map[kode.toUpperCase()] = r; rows.push(r);
  }
  dash.cells.forEach(function (c) {
    var r = map[String(c.kode || '').toUpperCase()];
    if (!r) return;
    r.total++;
    if (c.s && r[c.s] !== undefined) r[c.s]++; else r.belum++;
  });
  rows.sort(function (a, b) { return a.nama.localeCompare(b.nama, 'id'); });
  return { tanggal: dash.tanggal, rows: rows };
}

function listUser(token) {
  auth_(token, true);
  var d = sh_('Data_Guru').getDataRange().getValues(), o = [];
  for (var i = 1; i < d.length; i++) if (d[i][0]) o.push({ username: d[i][0], nama: d[i][2], role: d[i][3] || 'guru' });
  return o;
}

function tambahUser(token, u) {
  try {
    auth_(token, true);
    if (!u.username || !u.password || !u.nama) return { success: false, message: 'Semua kolom wajib diisi.' };
    var sheet = sh_('Data_Guru'), d = sheet.getDataRange().getValues();
    for (var i = 1; i < d.length; i++)
      if (String(d[i][0]).toLowerCase() == u.username.trim().toLowerCase()) return { success: false, message: 'Username sudah dipakai.' };
    sheet.appendRow([u.username.trim(), u.password, u.nama.trim(), u.role === 'admin' ? 'admin' : 'guru']);
    return { success: true, message: 'User ' + u.nama + ' berhasil ditambahkan.' };
  } catch (e) { return { success: false, message: e.message }; }
}

function hapusUser(token, username) {
  try {
    var me = auth_(token, true);
    if (String(username).toLowerCase() == String(me.username).toLowerCase()) return { success: false, message: 'Tidak bisa menghapus akun sendiri.' };
    var sheet = sh_('Data_Guru'), d = sheet.getDataRange().getValues();
    for (var i = 1; i < d.length; i++)
      if (d[i][0] == username) { sheet.deleteRow(i + 1); return { success: true, message: 'User dihapus.' }; }
    return { success: false, message: 'User tidak ditemukan.' };
  } catch (e) { return { success: false, message: e.message }; }
}

// ===== ADMIN: AKTIVITAS TERBARU =====
function getAktivitas(token) {
  auth_(token, true);
  var d = sh_('Log_Absensi').getDataRange().getValues(), o = [];
  for (var i = d.length - 1; i >= 1 && o.length < 8; i--) {
    var t = d[i][0];
    o.push({ jam: t instanceof Date ? Utilities.formatDate(t, TZ, 'dd/MM HH:mm') : '', piket: d[i][3], kelas: d[i][4],
      ke: d[i][5], guru: d[i][6], s: d[i][7] });
  }
  return o;
}

// ===== ADMIN: EDIT USER =====
// Username tidak bisa diubah (dipakai sebagai Kode_Guru di sheet Jadwal).
// Password kosong = tidak diubah.
function editUser(token, u) {
  try {
    var me = auth_(token, true);
    if (!u.nama || !String(u.nama).trim()) return { success: false, message: 'Nama wajib diisi.' };
    var role = u.role === 'admin' ? 'admin' : 'guru';
    var sheet = sh_('Data_Guru'), d = sheet.getDataRange().getValues(), row = -1, admins = 0;
    for (var i = 1; i < d.length; i++) {
      if (String(d[i][3] || '').toLowerCase() === 'admin') admins++;
      if (String(d[i][0]).toLowerCase() === String(u.username).toLowerCase()) row = i;
    }
    if (row < 0) return { success: false, message: 'User tidak ditemukan.' };
    var wasAdmin = String(d[row][3] || '').toLowerCase() === 'admin';
    if (wasAdmin && role !== 'admin' && admins <= 1) return { success: false, message: 'Tidak bisa menurunkan admin terakhir.' };
    if (String(d[row][0]).toLowerCase() === String(me.username).toLowerCase() && role !== 'admin')
      return { success: false, message: 'Tidak bisa menurunkan role akun sendiri.' };
    sheet.getRange(row + 1, 3).setValue(String(u.nama).trim());
    sheet.getRange(row + 1, 4).setValue(role);
    if (u.password) sheet.getRange(row + 1, 2).setValue(u.password);
    if (String(d[row][0]).toLowerCase() === String(me.username).toLowerCase()) { // perbarui sesi sendiri
      me.name = String(u.nama).trim();
      CacheService.getScriptCache().put(token, JSON.stringify(me), 21600);
    }
    return { success: true, message: 'Data ' + String(u.nama).trim() + ' berhasil diperbarui.' };
  } catch (e) { return { success: false, message: e.message }; }
}
